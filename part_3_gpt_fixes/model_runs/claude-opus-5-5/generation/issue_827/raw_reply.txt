```python
from qiskit import QuantumRegister, ClassicalRegister, QuantumCircuit
# FIX: qiskit.compile and qiskit.Aer no longer exist -> import transpile from qiskit and Aer from qiskit_aer, because transpile replaces compile and Aer now lives in qiskit_aer
from qiskit import transpile
from qiskit_aer import Aer
# FIX: qiskit.backends.jobstatus was removed -> import from qiskit.providers.jobstatus, because that is where JOB_FINAL_STATES is defined now
from qiskit.providers.jobstatus import JOB_FINAL_STATES

n_qubits = 5
qc_list = []
for i in range(n_qubits):
    qr = QuantumRegister(n_qubits)
    cr = ClassicalRegister(n_qubits)
    qc = QuantumCircuit(qr, cr)
    qc.x(qr[i])
    qc.measure(qr, cr)
    qc_list.append(qc)

backend = Aer.get_backend('qasm_simulator')

# FIX: compile() produced a qobj and was removed -> use transpile(), because backend.run() now takes transpiled circuits directly
qobj_list = [transpile(qc, backend) for qc in qc_list]
job_list = [backend.run(qobj) for qobj in qobj_list]

while job_list:
    for job in job_list:
        if job.status() in JOB_FINAL_STATES:
            job_list.remove(job)
            print(job.result().get_counts())
```
