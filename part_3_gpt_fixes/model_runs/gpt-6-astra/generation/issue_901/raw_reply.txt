```python
from qiskit.circuit import QuantumCircuit
from qiskit import transpile
# FIX: The translator lacks an S-to-T equivalence -> import the gate and session library, because exact basis translation needs a registered rule.
from qiskit.circuit.library import SGate
from qiskit.circuit.equivalence_library import SessionEquivalenceLibrary

qc = QuantumCircuit(2)
qc.h(0)
qc.s(0)
qc.cx(0, 1)

basis_gates = ["h", "t", "cx", "id"]
# FIX: S cannot be translated using the available rules -> register S = T T, because two T gates exactly implement S in the requested basis.
s_decomposition = QuantumCircuit(1)
s_decomposition.t(0)
s_decomposition.t(0)
SessionEquivalenceLibrary.add_equivalence(SGate(), s_decomposition)
qc_transpiled = transpile(qc, basis_gates=basis_gates)
```
