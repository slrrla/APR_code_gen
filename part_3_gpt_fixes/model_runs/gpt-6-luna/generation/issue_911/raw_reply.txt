```python
import qiskit
from qiskit.circuit.random import random_circuit

num_qubits = 3
depth = 1
max_operands = 2
qc = random_circuit(num_qubits, depth, max_operands=max_operands, seed=1)
qc = qc.decompose(reps=3)

for name_of_gate, qargs, cargs in qc.data:
    print("name of gate : ", name_of_gate)
    print("qargs : ", qargs, "\n")

    # FIX: Qubit attributes and indexing are not public APIs, and qargs[1] fails for single-qubit gates -> use find_bit for each argument, because it provides the qubit's register and local index.
    for qarg in qargs:
        location = qc.find_bit(qarg)
        print("register : ", location.registers[0][0])
        print("index : ", location.registers[0][1])
```
