```python
import numpy as np
from qiskit import QuantumCircuit
from qiskit.opflow import I, X, Y, Z

circuit = QuantumCircuit(1, name='R')
# FIX: appending the opflow expression creates a circuit instruction, not a gate -> append its matrix as a unitary, because UnitaryGate can be converted by to_gate()
circuit.unitary((0.5*I - 1j*np.sqrt(1-0.5**2)*Y).to_matrix(), [0])
circuit.to_gate()
```
