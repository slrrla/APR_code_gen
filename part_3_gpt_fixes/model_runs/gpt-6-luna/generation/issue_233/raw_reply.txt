```python
from qiskit.circuit.library import TwoLocal
import numpy as np

num_qubits = 2

rot = TwoLocal(
    int(np.sum(num_qubits)),
    "ry",
    [],
    reps=1,
    skip_final_rotation_layer=True,
    parameter_prefix="p",
)
var = TwoLocal(
    int(np.sum(num_qubits)),
    "ry",
    "cx",
    entanglement="linear",
    reps=1,
    skip_final_rotation_layer=True,
)

# FIX: num_parameters_settable() reflects the TwoLocal template, not the composed circuit -> use num_parameters, because it counts the circuit's actual unique parameters
print(rot.num_parameters)

rot.compose(var, inplace=True)

# FIX: num_parameters_settable() does not report parameters added by composition -> use num_parameters, because it counts the circuit's actual unique parameters
print(rot.num_parameters)
```
