from qiskit import QuantumCircuit, ClassicalRegister, QuantumRegister, transpile
from qiskit_aer import Aer

def oracle(circuit, register, marked_state):
    for i in range(len(marked_state)):
        if marked_state[i] == '1':
            circuit.x(register[i])
    circuit.cz(register[0], register[1])
    for i in range(len(marked_state)):
        if marked_state[i] == '1':
            circuit.x(register[i])

def grover_diffusion(circuit, register):
    circuit.h(register)
    circuit.x(register)
    circuit.h(register[1])
    circuit.cx(register[0], register[1])
    circuit.h(register[1])
    circuit.x(register)
    circuit.h(register)

def grover(marked_state):
    n = len(marked_state)
    cr = ClassicalRegister(n)
    # FIX: the circuit must use the same register passed to its gates -> create and reuse qr, because newly created registers are not part of the circuit
    qr = QuantumRegister(n)
    circuit = QuantumCircuit(qr, cr)
    print(qr)
    # FIX: gates must target qubits belonging to this circuit -> use qr, because a fresh QuantumRegister causes a CircuitError
    circuit.h(qr)
    num_iterations = int(round((2 ** n) ** 0.5))
    for i in range(num_iterations):
        # FIX: oracle and diffusion must use the circuit's register -> pass qr, because separate registers are not circuit qubits
        oracle(circuit, qr, marked_state)
        grover_diffusion(circuit, qr)
    # FIX: measurement must use the circuit's register -> measure qr, because a fresh register is not part of the circuit
    circuit.measure(qr, cr)
    backend = Aer.get_backend('qasm_simulator')
    # FIX: transpile returns a circuit, not a job -> run the transpiled circuit on the backend, because result() belongs to the returned job
    job = backend.run(transpile(circuit, backend))
    result = job.result()
    counts = result.get_counts()
    x = list(counts.keys())[0]
    return x

marked_state = '101'
result = grover(marked_state)
print(f"The marked state is {result}")
