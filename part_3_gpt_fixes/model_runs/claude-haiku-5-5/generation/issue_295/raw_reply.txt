# NO BUG: `circuit += oracle` is valid Qiskit (it is `QuantumCircuit.__iadd__`, which appends the oracle's instructions onto the same two qubits), and the Deutsch circuit logic (X and H on the ancilla, CNOT oracle for the balanced case, final H, measure qubit 0) is correct.

```python
from qiskit import QuantumCircuit, Aer, execute

def deutsch_circuit(oracle):
    circuit = QuantumCircuit(2, 1)
    circuit.x(1)
    circuit.h(0)
    circuit.h(1)
    circuit += oracle
    circuit.h(0)
    circuit.measure(0, 0)
    return circuit

def constant_oracle():
    circuit = QuantumCircuit(2)
    return circuit

def balanced_oracle():
    circuit = QuantumCircuit(2)
    circuit.cx(0, 1)
    return circuit

circuit = deutsch_circuit(constant_oracle())
backend = Aer.get_backend('qasm_simulator')
result = execute(circuit, backend).result()
print(result.get_counts())

circuit = deutsch_circuit(balanced_oracle())
backend = Aer.get_backend('qasm_simulator')
result = execute(circuit, backend).result()
print(result.get_counts())
```
