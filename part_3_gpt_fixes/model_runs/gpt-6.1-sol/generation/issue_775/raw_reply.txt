```python
# FIX: The QASM parser was not imported -> import QuantumCircuit, because execute requires a circuit rather than QASM text.
from qiskit import execute, Aer, QuantumCircuit

qasm_str = '''
OPENQASM 2.0;
include "qelib1.inc";
qreg q[2];
creg c[2];
h q[0];
cx q[0],q[1];
'''

backend = Aer.get_backend('qasm_simulator')
# FIX: execute received raw QASM text -> parse it into a QuantumCircuit, because execute accepts circuits, not strings.
job = execute(QuantumCircuit.from_qasm_str(qasm_str), backend)
result = job.result()
print(result)
```
