```python
from qiskit import QuantumCircuit, transpile
from qiskit.providers.fake_provider import FakeVigo
# FIX: CSPLayout is not a transpile layout method -> import the explicit pass and its supporting classes, because CSP must be run through a pass manager.
from qiskit.transpiler import CouplingMap, PassManager
from qiskit.transpiler.passes import CSPLayout

qc = QuantumCircuit(3)
qc.h(0)
qc.cx(0, 1)
qc.cx(1, 2)
qc.measure_all()

backend = FakeVigo()

# FIX: No CSP layout was computed -> run CSPLayout on the backend coupling map, because its resulting layout can be supplied to transpile.
pass_manager = PassManager(CSPLayout(CouplingMap(backend.configuration().coupling_map)))
pass_manager.run(qc)

transpiled = transpile(
    qc,
    backend=backend,
    optimization_level=3,
    # FIX: 'csp_layout' is an invalid layout method -> supply the computed CSP layout as initial_layout, because transpile accepts an explicit layout.
    initial_layout=pass_manager.property_set['layout']
)

print(transpiled)
```
