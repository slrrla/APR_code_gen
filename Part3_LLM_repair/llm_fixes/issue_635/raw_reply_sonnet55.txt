```python
from qiskit import QuantumRegister, ClassicalRegister, QuantumCircuit, execute, Aer

q = QuantumRegister(3)
c = ClassicalRegister(3)
qc = QuantumCircuit(q, c)

qc.measure(q, c)

# FIX: each X was conditioned on a single bit, so the two gates fired independently -> condition both X gates on the whole register value 3 (c[0]=1 and c[1]=1), because the register-level c_if makes both gates apply only when both bits are 1, as the question wants.
qc.x(q[0]).c_if(c, 3)
qc.x(q[1]).c_if(c, 3)

backend = Aer.get_backend('qasm_simulator')
job = execute(qc, backend, shots=1024)
result = job.result()
print(result.get_counts(qc))
```
