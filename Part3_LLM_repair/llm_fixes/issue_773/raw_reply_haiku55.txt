```python
from qiskit import QuantumCircuit, execute, Aer
# FIX: duplicate job_monitor import from qiskit.providers.ibmq.job shadowed the working one and only works with IBM Quantum jobs -> removed the ibmq import so job_monitor is the qiskit.tools.monitor version, because it is the one that works with the local Aer simulator job.
from qiskit.tools.monitor import job_monitor

qc = QuantumCircuit(1, 1)
qc.h(0)
qc.measure(0, 0)

backend = Aer.get_backend('qasm_simulator')
job = execute(qc, backend)

job_monitor(job)
```
