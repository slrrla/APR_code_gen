```python
import qiskit
from qiskit import QuantumCircuit, QuantumRegister

qr = QuantumRegister(2, 'q')
qc = QuantumCircuit(qr)
qc.h(0)
qc.cx(0, 1)

for instruction, qargs, cargs in qc.data:
    qbit: qiskit.circuit.Qubit = qargs[0]
    # FIX: qbit.register used the deprecated Bit-to-Register back-reference -> replaced with qc.find_bit(qbit).registers, because find_bit looks up the bit's registers through the circuit instead
    print(qc.find_bit(qbit).registers)
    # FIX: qbit.index used the deprecated Bit-to-Register back-reference -> replaced with qc.find_bit(qbit).index, because find_bit returns the bit's index within the circuit without relying on the back-reference
    print(qc.find_bit(qbit).index)
```
