```python
import qiskit
from qiskit import QuantumCircuit, QuantumRegister

qr = QuantumRegister(2, 'q')
qc = QuantumCircuit(qr)
qc.h(0)
qc.cx(0, 1)

for instruction, qargs, cargs in qc.data:
    qbit: qiskit.circuit.Qubit = qargs[0]
    # FIX: Deprecated Bit back-references -> find the containing register and bit position from the register, because this avoids deprecated accessors.
    register = next(register for register in qc.qregs if qbit in register)
    print(register)
    print(register.index(qbit))
```
