The pulse simulator does not support `set_frequency`, so the job errors out. The fix follows the workaround from the Aer issue: drop the frequency change and modulate the pulse samples instead.

- I sample the Gaussian into an explicit `Waveform`.
- I multiply its samples by `exp(i·2π(v−w)·t)`, where `t = n·dt`, `v` is the target frequency and `w` is the drive channel's default LO frequency (`qubit_freq_est`).

The sign of the exponent follows the Qiskit Pulse convention, signal = Re[d(t)·e^{i2πft}]. I have not run this, and I am not certain the Aer pulse simulator uses the same sign. If it uses the opposite one, the spectroscopy peak lands at the mirrored offset from the LO frequency rather than at the requested frequency. In that case, flip the sign of the exponent.

```python
from qiskit import pulse, assemble
from qiskit.pulse import Play, Schedule, DriveChannel, Gaussian, AcquireChannel, Waveform
from qiskit.providers.aer import PulseSimulator
from qiskit.test.mock import FakeArmonk
import numpy as np

job_params = {
    'meas_level': 1,
    'meas_return': 'avg',
    'shots': 512
}
spec_range = 0.300
num_spec01_freqs = 71
GHz = 1.0e9

def get_exc_chans(gv):
    return [AcquireChannel(i) for i in range(gv['backend_config'].n_qubits)]

def get_spec01_freqs(center_freqs, qubit):
    center_freq = round(center_freqs[qubit], -8)
    return np.linspace(center_freq/GHz - spec_range/2,
                        center_freq/GHz + spec_range/2,
                        num_spec01_freqs)

backend_real = FakeArmonk()
backend = PulseSimulator.from_backend(backend_real)
qubit = 0
backend_config = backend.configuration()
exc_chans = get_exc_chans(globals())
dt = backend_config.dt
backend_defaults = backend.defaults()
center_frequency = backend_defaults.qubit_freq_est
inst_sched_map = backend_defaults.instruction_schedule_map

meas = inst_sched_map.get('measure', qubits=[qubit])

drive_amp = 0.25
drive_duration = inst_sched_map.get('x', qubits=[qubit]).duration
drive_sigma = drive_duration // 4
spec_pulse = Gaussian(duration=drive_duration, amp=drive_amp, sigma=drive_sigma,
                       name=f"Spec drive amplitude = {drive_amp}")

spec_freqs_GHz = get_spec01_freqs(center_frequency, qubit)

spec01_scheds = []
for freq in spec_freqs_GHz:
    with pulse.build(name="Spec Pulse at %.3f GHz" % freq) as spec01_sched:
        with pulse.align_sequential():
            # FIX: set_frequency is unsupported by PulseSimulator (job errors) -> play an explicit Waveform modulated by exp(i*2*pi*(v-w)*t), because this shifts the drive from the LO frequency w to the target frequency v
            pulse.play(Waveform(spec_pulse.get_waveform().samples * np.exp(1j*2*np.pi*(freq*GHz - center_frequency[qubit])*np.arange(drive_duration)*dt)), DriveChannel(qubit))
            pulse.call(meas)
    spec01_scheds.append(spec01_sched)

qobj = assemble(spec01_scheds, backend=backend, **job_params)
spec01_job = backend.run(qobj)
print(spec01_job.result())
```
