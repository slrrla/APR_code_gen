from qiskit import QuantumCircuit, ClassicalRegister, QuantumRegister, transpile
from qiskit_aer import Aer

def oracle(circuit, register, marked_state):
    for i in range(len(marked_state)):
        if marked_state[i] == '1':
            circuit.x(register[i])
    circuit.cz(register[0], register[1])
    for i in range(len(marked_state)):
        if marked_state[i] == '1':
            circuit.x(register[i])

def grover_diffusion(circuit, register):
    circuit.h(register)
    circuit.x(register)
    circuit.h(register[1])
    circuit.cx(register[0], register[1])
    circuit.h(register[1])
    circuit.x(register)
    circuit.h(register)

def grover(marked_state):
    n = len(marked_state)
    cr = ClassicalRegister(n)
    # FIX: The circuit's register must be reused -> keep its qubits in qr, because newly created registers contain different qubits.
    qr = QuantumRegister(n)
    circuit = QuantumCircuit(qr, cr)
    print(qr)
    # FIX: A new QuantumRegister is not part of the circuit -> apply gates to qr, because it is the register attached to the circuit.
    circuit.h(qr)
    num_iterations = int(round((2 ** n) ** 0.5))
    for i in range(num_iterations):
        # FIX: New registers are not circuit qubits -> pass qr to the operations, because it is the register attached to the circuit.
        oracle(circuit, qr, marked_state)
        grover_diffusion(circuit, qr)
    # FIX: Measuring a newly created register causes a CircuitError -> measure qr, because it belongs to the circuit.
    circuit.measure(qr, cr)
    backend = Aer.get_backend('qasm_simulator')
    # FIX: transpile returns a circuit, not a job -> run the transpiled circuit on the backend, because result() belongs to the job.
    job = backend.run(transpile(circuit, backend))
    result = job.result()
    counts = result.get_counts()
    x = list(counts.keys())[0]
    return x

marked_state = '101'
result = grover(marked_state)
print(f"The marked state is {result}")
