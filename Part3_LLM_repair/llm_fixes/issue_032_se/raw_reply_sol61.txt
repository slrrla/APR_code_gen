# FIX: iSWAP needs a gate instance rather than an unsupported layer name -> import iSwapGate, because TwoLocal accepts gate objects.
from qiskit.circuit.library import TwoLocal, iSwapGate

num_spin_orbitals = 4

ansatz = TwoLocal(
    num_spin_orbitals,
    ['ry', 'rz'],
    # FIX: 'iswap' is not recognized as a layer name -> pass iSwapGate(), because it supplies the intended entangler directly.
    entanglement_blocks=iSwapGate(),
    entanglement='linear'
)

print(ansatz.decompose())
