```python
from qiskit.circuit import QuantumRegister, AncillaRegister, QuantumCircuit

qr1 = QuantumRegister(4)
anc1 = AncillaRegister(2)
qc1 = QuantumCircuit(qr1, anc1)
qc1.ccx(qr1[0], qr1[1], anc1[0])
qc1.ccx(qr1[2], anc1[0], anc1[1])
qc1.cx(anc1[1], qr1[3])
qc1.ccx(qr1[2], anc1[0], anc1[1])
qc1.ccx(qr1[0], qr1[1], anc1[0])

# FIX: the target has no room for the subcircuit's two ancillas -> add an ancilla register, because compose requires the target to have a qubit for every source qubit
circ = QuantumCircuit(QuantumRegister(4), AncillaRegister(2))
circ.h([0, 1, 2, 3])
circ.barrier()

# FIX: only four target qubits were mapped for a six-qubit subcircuit -> map all six qubits, because compose needs one target qubit per source qubit
circ.compose(qc1, [0, 1, 2, 3, 4, 5], inplace=True)
```
