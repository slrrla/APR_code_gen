```python
import numpy as np
from qiskit import QuantumCircuit
from qiskit.opflow import I, X, Y, Z

circuit = QuantumCircuit(1, name='R')
# FIX: appended an opflow operator, which becomes a non-Gate Instruction -> convert it to its matrix and add it with circuit.unitary, because that adds a UnitaryGate (a real Gate) so to_gate() works
circuit.unitary((0.5*I - 1j*np.sqrt(1-0.5**2)*Y).to_matrix(), [0])
circuit.to_gate()
```
