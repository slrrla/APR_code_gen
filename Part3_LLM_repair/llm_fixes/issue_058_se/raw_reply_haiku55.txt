The statevector simulator rejects the custom `U$_s$` gate because it is not in its basis gates. Transpiling the circuit for the backend decomposes that gate into supported instructions before the run.

```python
# FIX: Simulator rejects the custom "U$_s$" gate as an invalid instruction -> added transpile to the qiskit import, because it is needed to decompose the custom gate into backend-supported gates
from qiskit import QuantumCircuit, QuantumRegister, Aer, transpile

def diffuser(nqubits):
    qc = QuantumCircuit(nqubits)
    for qubit in range(nqubits):
        qc.h(qubit)
    for qubit in range(nqubits):
        qc.x(qubit)
    qc.h(nqubits-1)
    qc.mct(list(range(nqubits-1)), nqubits-1)
    qc.h(nqubits-1)
    for qubit in range(nqubits):
        qc.x(qubit)
    for qubit in range(nqubits):
        qc.h(qubit)
    U_s = qc.to_gate()
    U_s.name = "U$_s$"
    return U_s

psi = QuantumRegister(3, 'psi')

backend = Aer.get_backend("statevector_simulator")
circuit = QuantumCircuit(psi)
circuit.append(diffuser(3), psi)

# FIX: Circuit was passed to the backend without transpiling -> transpile it for the backend before running, because this decomposes the custom gate into supported instructions
job = backend.run(transpile(circuit, backend))

v = job.result().get_statevector(circuit)
```
