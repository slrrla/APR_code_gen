```python
from qiskit import QuantumCircuit
# FIX: BasicProvider does not support Aer-specific parallel options -> import AerSimulator, because it provides those options.
from qiskit_aer import AerSimulator

qc_combine = QuantumCircuit(2)
qc_combine.h(0)
qc_combine.cx(0, 1)
qc_combine.measure_all()

shots_used = 1024

# FIX: BasicSimulator rejects the parallel options below -> use AerSimulator, because it supports them.
backend = AerSimulator()

backend.set_options(
    max_parallel_threads=0,
    max_parallel_experiments=0,
    max_parallel_shots=42,
    statevector_parallel_threshold=16
)

job = backend.run(qc_combine, shots=shots_used)
result = job.result()
counts = result.get_counts(qc_combine)
print(counts)
```
