```
from qiskit import IBMQ if IBMQ.active_account() is None: IBMQ.load_account() provider = IBMQ.get_provider(hub = 'ibm-q') backend = provider.get_backend('ibmq_quito') print('Status:') print(' Operational: ', backend.status().operational) print(' Pending jobs:', backend.status().pending_jobs) print(' Status message:', backend.status().status_msg)
```
