```python
import numpy as np
from qiskit import QuantumCircuit
from qiskit.opflow import I, X, Y, Z

circuit = QuantumCircuit(1, name='R')
# FIX: appending an opflow expression creates a non-gate circuit instruction -> append its matrix as a unitary gate, because to_gate() requires every instruction to be a gate
circuit.unitary((0.5*I - 1j*np.sqrt(1-0.5**2)*Y).to_matrix(), [0])
circuit.to_gate()
```
