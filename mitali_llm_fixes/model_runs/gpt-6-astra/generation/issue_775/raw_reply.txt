# FIX: OpenQASM text needs parsing -> import QuantumCircuit, because execute expects a circuit rather than source text
from qiskit import execute, Aer, QuantumCircuit

qasm_str = '''
OPENQASM 2.0;
include "qelib1.inc";
qreg q[2];
creg c[2];
h q[0];
cx q[0],q[1];
'''

backend = Aer.get_backend('qasm_simulator')
# FIX: execute received raw OpenQASM text -> parse it into a QuantumCircuit, because execute cannot run a source string directly
job = execute(QuantumCircuit.from_qasm_str(qasm_str), backend)
result = job.result()
print(result)
