```python
from qiskit import QuantumCircuit, transpile

lam = 0.5

qc = QuantumCircuit(1)
# FIX: u1 is no longer supported -> use the equivalent p gate, because transpilation preserves its phase via the circuit's global_phase.
qc.p(lam, 0)

transpiled = transpile(qc, basis_gates=['rx', 'ry', 'rz'])
print(transpiled)

```
