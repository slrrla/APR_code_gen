```python
from qiskit_aer import AerSimulator
from qiskit import QuantumCircuit, transpile

circ = QuantumCircuit(2)
circ.h(0)
circ.cx(0, 1)
# FIX: The circuit did not save its unitary -> add a save instruction, because Aer only returns saved quantities.
circ.save_unitary()

# FIX: The default simulator method does not compute unitary matrices -> select the unitary method, because it supports unitary simulation.
simulator = AerSimulator(method="unitary")
circ = transpile(circ, simulator)

result = simulator.run(circ).result()
unitary = result.get_unitary(circ)
print("Circuit unitary:\n", unitary)
```
