```python
from qiskit import QuantumRegister, ClassicalRegister, QuantumCircuit
# FIX: compile is deprecated -> import transpile and assemble, because these replace its two steps.
from qiskit import Aer
from qiskit.compiler import transpile, assemble
from qiskit.backends.jobstatus import JOB_FINAL_STATES

n_qubits = 5
qc_list = []
for i in range(n_qubits):
    qr = QuantumRegister(n_qubits)
    cr = ClassicalRegister(n_qubits)
    qc = QuantumCircuit(qr, cr)
    qc.x(qr[i])
    qc.measure(qr, cr)
    qc_list.append(qc)

backend = Aer.get_backend('qasm_simulator')

# FIX: compile combined transpilation and assembly -> call both replacements, because backend.run needs an assembled Qobj.
qobj_list = [assemble(transpile(qc, backend), backend) for qc in qc_list]
job_list = [backend.run(qobj) for qobj in qobj_list]

while job_list:
    for job in job_list:
        if job.status() in JOB_FINAL_STATES:
            job_list.remove(job)
            print(job.result().get_counts())
```
