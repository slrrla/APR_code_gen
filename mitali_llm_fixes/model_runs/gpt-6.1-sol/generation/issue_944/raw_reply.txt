import numpy as np
from qiskit.quantum_info import Statevector
# FIX: StateFn composition cannot form this projector -> import MatrixOp, because it represents matrix operators in opflow.
from qiskit.opflow import Plus, MatrixOp
from qiskit.algorithms import NumPyEigensolver

# FIX: Plus @ ~Plus is unsupported StateFn composition -> construct the outer product, because it yields the projector |+><+|.
proj = MatrixOp(np.outer(Plus.to_matrix(), Plus.to_matrix().conj()))

plus = Statevector([1/np.sqrt(2), 1/np.sqrt(2)]).to_operator()
solver = NumPyEigensolver()
# FIX: quantum_info.Operator lacks to_spmatrix -> wrap it in MatrixOp, because NumPyEigensolver requires an opflow operator.
spectrum = solver.compute_eigenvalues(MatrixOp(plus))
