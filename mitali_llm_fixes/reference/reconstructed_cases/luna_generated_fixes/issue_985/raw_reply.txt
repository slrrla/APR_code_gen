```python
from qiskit import QuantumCircuit, transpile

lam = 0.5

qc = QuantumCircuit(1)
# FIX: u1 cannot be unrolled into determinant-one rotation gates while preserving phase -> use Rz and add the equivalent global phase
qc.rz(lam, 0)
qc.global_phase = lam / 2

transpiled = transpile(qc, basis_gates=['rx', 'ry', 'rz'])
print(transpiled)
```
