from qiskit import QuantumRegister, QuantumCircuit, AncillaRegister, transpile from qiskit_aer import StatevectorSimulator import numpy as np reg1 = QuantumRegister(2, 'psi') reg2 = AncillaRegister(2, 'anc') circ = QuantumCircuit(reg1, reg2) circ.h(reg1[0]) circ.h(reg1[1]) circ.prepare_state(1/np.sqrt(2)*np.array([1,0,0,1]),reg2) circ.cry(np.pi/2.2, control_qubit=reg2[1], target_qubit=reg1[1]) circ.cry(np.pi/1.5, control_qubit=reg2[0], target_qubit=reg1[0]) circ.prepare_state(1/np.sqrt(2)*np.array([1,0,0,1]),reg2).inverse() simulator = StatevectorSimulator() job = transpile(circ, simulator) # execute circuit on qasm simulator result = simulator.run(job,shots=128).result() # results from the job out_state = result.get_statevector() print(out_state) circ.draw('mpl')

from qiskit import QuantumRegister, QuantumCircuit, AncillaRegister from qiskit_aer import AerSimulator reg1 = QuantumRegister(1, 'psi') reg2 = AncillaRegister(1, 'anc') circ = QuantumCircuit(reg1, reg2) circ.h(reg1) circ.x(reg2) circ.cz(reg1,reg2) circ.h(reg1) circ.x(reg2) circ.save_density_matrix(reg1) ρ = AerSimulator().run(circ).result().data()['density_matrix'] ψ = ρ.to_statevector() print(ψ)
```
